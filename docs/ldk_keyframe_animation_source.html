typedef struct LDKKeyFrameAnimationSource{
  LDKAssetKeyframeAnimation *animations;
  u32 animation_count;
  u32 animation_capacity;
  i32 current_animation;
  bool play_on_start;
  bool loop;
  float speed;
  float time;
  bool playing;
  bool runtime_started;
  bool restart_pending;
}LDKKeyFrameAnimationSource;
